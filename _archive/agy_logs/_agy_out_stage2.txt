### 1. Design Architetturale dello Smoke Test Stage 2 (FitNets)

1. **Topologia e Moduli**: Il modello integra Gemma 4 E2B (quantizzato 4-bit, frozen), una convoluzione causale 1D (`in=3072, out=3072, kernel=3`, left-pad `(2, 0)`, frozen), il modulo di distillazione `ht_adapter` (`Linear(3072, 2048) + LayerNorm(2048)` in FP32, **unico componente trainable**), e il `subtalker` Qwen3-TTS (1.7B code predictor, frozen in BF16).

        import json

        import torch
        import torch.nn as nn
        import torch.nn.functional as F
        from transformers import AutoTokenizer

        # ----------------------------------------------------------------------
        # 1. Hyperparameters & Configuration
        # ----------------------------------------------------------------------
        SMOKE_BATCHES = int(os.environ.get("PHASE2_SMOKE_BATCHES", "4"))
        BATCH_SIZE = int(os.environ.get("STAGE2_BATCH_SIZE", "8"))
        GRAD_ACCUM = int(os.environ.get("STAGE2_GRAD_ACCUM", "2"))
        LR = float(os.environ.get("STAGE2_LR", "1.5e-4"))
        LAMBDA_COUPLING = float(os.environ.get("STAGE2_LAMBDA_COUPLING", "0.2"))
        WEIGHT_DECAY = float(os.environ.get("STAGE2_WEIGHT_DECAY", "1e-2"))
        GRAD_CLIP = float(os.environ.get("STAGE2_GRAD_CLIP", "1.0"))

        TEXT_BLOCK_LEN = 48
        AUDIO_FEAT_START = 2 + TEXT_BLOCK_LEN  # 50
        FPF = 2                                # 2 audio features per 12.5Hz agent frame
        N_AGENT = 64                           # 64 agent frames per window
        DEVICE = torch.device("cuda:0" if torch.cuda.is_available() else "cpu")

        BOS_ID = 2
        BOA_ID = 256000
        AUDIO_TOKEN_ID = 258881
        EOA_ID = 258883

        print(f"[Stage2 Smoke] Config: batches={SMOKE_BATCHES}, batch_size={BATCH_SIZE}, "
              f"grad_accum={GRAD_ACCUM}, lr={LR}, lambda_coupling={LAMBDA_COUPLING}", flush=True)

        # ----------------------------------------------------------------------
        # 2. Local Slim Architecture Definitions
        # ----------------------------------------------------------------------
        class CausalConv1d(nn.Module):
            def __init__(self, in_channels=3072, out_channels=3072, kernel_size=3):
                super().__init__()
                self.conv = nn.Conv1d(in_channels, out_channels, kernel_size=kernel_size, padding=0)
                self.kernel_size = kernel_size

            def forward(self, x):
                # x: (B, T, C) -> transpose to (B, C, T)
                x_t = x.transpose(1, 2)
                x_padded = F.pad(x_t, (self.kernel_size - 1, 0))
                out = self.conv(x_padded)
                return out.transpose(1, 2)

        class Stage2SlimModel(nn.Module):
            def __init__(self, base_model, subtalker, hidden_size=3072, talker_hidden=2048):
                super().__init__()
                self.base_model = base_model
                self.causal_conv = CausalConv1d(hidden_size, hidden_size, kernel_size=3)
                self.ht_adapter = nn.Sequential(
                    nn.Linear(hidden_size, talker_hidden),
                    nn.LayerNorm(talker_hidden)
                )
                self.subtalker = subtalker

            def forward_adapter(self, agent_hidden):
                # agent_hidden: (B, 64, 3072) fp32 -> adapted -> ht_adapter -> (B, 64, 2048) fp32
                adapted = self.causal_conv(agent_hidden)
                h_t_pred = self.ht_adapter(adapted)
                return h_t_pred

        # ----------------------------------------------------------------------
        # 3. Load Gemma 4 E2B Base Model (Frozen, 4-bit) & Tokenizer
        # ----------------------------------------------------------------------
        print("[Stage2 Smoke] Loading Gemma 4 E2B base model (4-bit)...", flush=True)
        try:
            from unsloth import FastLanguageModel
            base_model, tokenizer = FastLanguageModel.from_pretrained(
                model_name="unsloth/gemma-4-E2B-it",
                max_seq_length=2048,
                load_in_4bit=True,
                device_map="cuda:0",
            )
        except Exception as e_unsloth:
            print(f"[Stage2 Smoke] FastLanguageModel load failed ({e_unsloth}), trying AutoModelForCausalLM...", flush=True)
            from transformers import AutoModelForCausalLM, BitsAndBytesConfig
            bnb_config = BitsAndBytesConfig(
                load_in_4bit=True,
                bnb_4bit_quant_type="nf4",
                bnb_4bit_compute_dtype=torch.bfloat16,
                bnb_4bit_use_double_quant=True,
            )
            base_model = AutoModelForCausalLM.from_pretrained(
                "unsloth/gemma-4-E2B-it",
                quantization_config=bnb_config,
                device_map="cuda:0",
                torch_dtype=torch.bfloat16,
            )
            tokenizer = AutoTokenizer.from_pretrained("unsloth/gemma-4-E2B-it")

        # T4 Dtype Fix: cast all LayerNorm modules to fp32 to prevent runtime mixed-precision error
        norm_count = 0
        for name, module in base_model.named_modules():
            if "norm" in name.lower():
                module.to(torch.float32)
                norm_count += 1
        print(f"[Stage2 Smoke] Converted {norm_count} norm modules to fp32 for T4 stability.", flush=True)

        # Freeze base model completely
        base_model.eval()
        for p in base_model.parameters():
            p.requires_grad = False

        # Ensure subtalker is frozen
        if "sub_talker" in globals() and sub_talker is not None:
            subtalker_instance = sub_talker
        elif "subtalker" in globals() and subtalker is not None:
            subtalker_instance = subtalker
        else:
            raise RuntimeError("Subtalker instance not found in globals. Ensure subtalker is loaded prior to STAGE2.")

        subtalker_instance.eval()
        for p in subtalker_instance.parameters():
            p.requires_grad = False

        # Instantiate Stage 2 Model
        stage2_model = Stage2SlimModel(base_model, subtalker_instance).to(DEVICE)

        # Freeze causal conv; keep only ht_adapter trainable
        stage2_model.causal_conv.requires_grad_(False)
        stage2_model.ht_adapter.train()
        for p in stage2_model.ht_adapter.parameters():
            p.requires_grad = True

        # ----------------------------------------------------------------------
        # 4. Optimizer Setup (AdamW ONLY on ht_adapter)
        # ----------------------------------------------------------------------
        trainable_params = [p for p in stage2_model.ht_adapter.parameters() if p.requires_grad]
        trainable_param_count = sum(p.numel() for p in trainable_params)
        print(f"[Stage2 Smoke] Trainable parameters in ht_adapter: {trainable_param_count:,}", flush=True)

        optimizer = torch.optim.AdamW(
            trainable_params,
            lr=LR,
            betas=(0.9, 0.98),
            eps=1e-6,
            weight_decay=WEIGHT_DECAY,
        )

        # ----------------------------------------------------------------------
        # 5. Helper Functions for Input Construction & Loss
        # ----------------------------------------------------------------------
        def build_gemma_input_ids(text_list, tokenizer_inst, dev):
            pad_id = tokenizer_inst.pad_token_id if tokenizer_inst.pad_token_id is not None else 0
            batch_input_ids = []
            for txt in text_list:
                toks = tokenizer_inst.encode(str(txt) if txt else "", add_special_tokens=False)
                if len(toks) > TEXT_BLOCK_LEN:
                    toks = toks[-TEXT_BLOCK_LEN:]
                elif len(toks) < TEXT_BLOCK_LEN:
                    toks = [pad_id] * (TEXT_BLOCK_LEN - len(toks)) + toks
                seq = [BOS_ID] + toks + [BOA_ID] + [AUDIO_TOKEN_ID] * (N_AGENT * FPF) + [EOA_ID]
                batch_input_ids.append(seq)
            return torch.tensor(batch_input_ids, dtype=torch.long, device=dev)

        total_windows_needed = SMOKE_BATCHES * BATCH_SIZE * GRAD_ACCUM
        available_wins = len(train_wins)

        step_logs = []
        win_cursor = 0
        optimizer.zero_grad()

        for step in range(1, SMOKE_BATCHES + 1):
            t0 = time.time()
            accum_loss = 0.0
            accum_latent = 0.0
            accum_coupling = 0.0
            accum_cossim = 0.0

            for micro_step in range(GRAD_ACCUM):
                # Slice batch indices
                batch_wins = [train_wins[(win_cursor + i) % available_wins] for i in range(BATCH_SIZE)]
                win_cursor = (win_cursor + BATCH_SIZE) % available_wins

                # Gather batch data
                items = [ds_train[w["idx"]] for w in batch_wins]
                texts = [text_by_window[w["idx"]] for w in batch_wins]
                n_reals = [w["n_real"] for w in batch_wins]

                # 1. Target hidden states h_t from teacher
                ht_targets_np = []
                for it, nr, txt in zip(items, n_reals, texts):
                    ht_item = extract_ht(it, nr, txt, return_shifted=False)
                    ht_targets_np.append(ht_item)
                ht_target = torch.from_numpy(np.stack(ht_targets_np, axis=0)).to(DEVICE, dtype=torch.float32)

                # 2. Audio targets & active masks
                audio_targets_list = [torch.from_numpy(it["audio_targets"]) if isinstance(it["audio_targets"], np.ndarray) else it["audio_targets"] for it in items]
                audio_targets = torch.stack(audio_targets_list, dim=0).to(DEVICE, dtype=torch.long)

                active_masks_list = []
                for nr in n_reals:
                    m = torch.zeros(N_AGENT, dtype=torch.float32)
                    m[:min(nr, N_AGENT)] = 1.0
                    active_masks_list.append(m)
                active_mask = torch.stack(active_masks_list, dim=0).to(DEVICE)  # (B, 64)

                # 3. Gemma Forward under torch.no_grad() (Detached representation, saves VRAM)
                input_ids = build_gemma_input_ids(texts, tokenizer, DEVICE)
                with torch.no_grad():

                    last_hidden = gemma_out.hidden_states[-1]  # (B, 179, 3072)

                    # Slicing: audio_feat_start=50 -> indices 51:178:2 gives exactly 64 frames
                    start_idx = AUDIO_FEAT_START + FPF - 1
                    end_idx = AUDIO_FEAT_START + FPF * N_AGENT
                    agent_hidden = last_hidden[:, start_idx:end_idx:FPF, :].to(torch.float32).detach()

                # 4. Trainable Adapter Forward (Builds autograd graph from agent_hidden)
                h_t_pred = stage2_model.forward_adapter(agent_hidden)  # (B, 64, 2048) fp32

                # 5. FitNets Latent Loss (Cosine + 0.5 * MSE on active frames)
                mask_sum = active_mask.sum() + 1e-8
                cos_sim_matrix = F.cosine_similarity(h_t_pred, ht_target, dim=-1)  # (B, 64)
                cos_loss = 1.0 - (cos_sim_matrix * active_mask).sum() / mask_sum
                mean_cos_sim = (cos_sim_matrix * active_mask).sum() / mask_sum

                mse_matrix = F.mse_loss(h_t_pred, ht_target, reduction="none").mean(dim=-1)  # (B, 64)
                mse_loss = (mse_matrix * active_mask).sum() / mask_sum

                L_latent = cos_loss + 0.5 * mse_loss

                # 6. Coupling Loss via Sub-talker (Cast h_t to bf16)
                h_t_sub = h_t_pred.to(torch.bfloat16)
                sub_logits = subtalker_instance.forward_sub_talker_finetune(h_t_sub, audio_targets)
                L_coupling = subtalker_instance.coupling_loss(sub_logits, audio_targets, frame_weights=active_mask)

                # 7. Total Micro-step Loss & Backward
                micro_loss = (L_latent + LAMBDA_COUPLING * L_coupling) / GRAD_ACCUM
                micro_loss.backward()

                accum_loss += micro_loss.item() * GRAD_ACCUM
                accum_latent += L_latent.item()
                accum_coupling += L_coupling.item()
                accum_cossim += mean_cos_sim.item()

            # Optimizer Step & Gradient Check
            grad_norm = torch.nn.utils.clip_grad_norm_(trainable_params, max_norm=GRAD_CLIP)
            optimizer.step()
            optimizer.zero_grad()

            # Check that frozen models have zero gradients
            base_has_grad = any(p.grad is not None and (p.grad != 0).any() for p in base_model.parameters())
            sub_has_grad = any(p.grad is not None and (p.grad != 0).any() for p in subtalker_instance.parameters())

            step_time = time.time() - t0
            avg_loss = accum_loss / GRAD_ACCUM
            avg_latent = accum_latent / GRAD_ACCUM
            avg_coupling = accum_coupling / GRAD_ACCUM
            avg_cossim = accum_cossim / GRAD_ACCUM

            step_info = {
                "step": step,
                "loss": round(float(avg_loss), 4),
                "latent_loss": round(float(avg_latent), 4),
                "coupling_loss": round(float(avg_coupling), 4),
                "cos_sim": round(float(avg_cossim), 4),
                "grad_norm": round(float(grad_norm.item() if hasattr(grad_norm, "item") else grad_norm), 4),
                "base_grad_zero": not base_has_grad,
                "sub_grad_zero": not sub_has_grad,
                "step_time_sec": round(float(step_time), 2),
            }
            step_logs.append(step_info)

            print(f"[Stage2 Smoke] Step {step}/{SMOKE_BATCHES} | Loss: {avg_loss:.4f} "
                  f"(Latent: {avg_latent:.4f}, Coupling: {avg_coupling:.4f}) | "
                  f"CosSim: {avg_cossim:.4f} | GradNorm: {step_info['grad_norm']:.4f} | "
                  f"BaseGradZero: {not base_has_grad} | SubGradZero: {not sub_has_grad} | "
                  f"Time: {step_time:.2f}s", flush=True)

        # ----------------------------------------------------------------------

        ckpt_path = os.path.join("checkpoints", "ht_adapter_stage2_smoke.pt")
        torch.save(stage2_model.ht_adapter.state_dict(), ckpt_path)
        print(f"[Stage2 Smoke] Adapter checkpoint successfully saved to: {ckpt_path}", flush=True)

        # 8. Summary Report Generation
        # ----------------------------------------------------------------------
        report = {
            "status": "PASS",
            "phase": "STAGE2_FITNETS_SMOKE",
            "batches_executed": SMOKE_BATCHES,
            "trainable_parameters": trainable_param_count,
            "base_frozen_verified": all(s["base_grad_zero"] for s in step_logs),

        report_path = os.path.join(output_dir, "stage2_smoke_report.json")

        print(f"[Stage2 Smoke] Execution report saved to: {report_path}", flush=True)
        print(">>> [STAGE 2 SMOKE TEST] COMPLETED SUCCESSFULLY!", flush=True)

    except Exception as e:
        print("[Stage2 Smoke] ERROR during Stage 2 Smoke Test execution!", flush=True)

### 5. Rischi e Note Tecniche (VRAM, Dtype, Slicing)

1. **Budget VRAM su T4 (16GB)**: Con Teacher Qwen3 (3.5GB), Gemma 4-bit (3.8GB) e Sub-talker (1.3GB), l'allocazione statica è ~8.6GB. Il forward di Gemma eseguito in `torch.no_grad()` con `agent_hidden.detach()` azzera i buffer delle attivazioni intermedie del base model, mantenendo il picco VRAM in backward stabilmente a ~10-11GB.
2. **Patch Dtype LayerNorm T4**: I layer di normalizzazione del base model vengono convertiti a FP32 (`module.to(torch.float32)` per ogni modulo `"norm"`), prevenendo `RuntimeError: expected scalar type Float but found BFloat16` durante il forward su GPU Turing.
3. **Bridge di Precisione Sub-Talker**: L'adapter opera in FP32 per stabilità di convergenza. All'ingresso di `subtalker.forward_sub_talker_finetune`, $h_t$ viene castato al volo a `torch.bfloat16` (`h_t_sub = h_t_pred.to(torch.bfloat16)`): PyTorch retropropaga correttamente il gradiente attraverso il cast e attraverso gli strati frozen del sub-talker verso `ht_adapter`.
4. **Allineamento Indici Slicing**: Con sequenza standard di 179 token e `AUDIO_FEAT_START = 50`, lo slicing `51:178:2` estrae esattamente 64 vettori temporali, ciascuno posizionato sulla feature di destra della coppia 25Hz corrispondente al frame agente a 12.5Hz.
